### 3. API-Client erstellen (lib/resellerinterface.ts)

```typescript
interface DomainCheckResult {
  available: boolean;
  domain: string;
  price?: number;
  suggestions?: string[];
}

export async function checkDomainAvailability(
  domain: string
): Promise<DomainCheckResult> {
  const response = await fetch(
    `${process.env.RESELLERINTERFACE_API_URL}/domain/check`,
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Basic ${Buffer.from(
          `${process.env.RESELLERINTERFACE_API_USERNAME}:${process.env.RESELLERINTERFACE_API_PASSWORD}`
        ).toString('base64')}`
      },
      body: JSON.stringify({
        domain: domain,
        reseller_id: process.env.RESELLERINTERFACE_RESELLER_ID
      })
    }
  );
  
  return await response.json();
}

export async function getDomainSuggestions(
  domainName: string,
  tlds: string[] = ['com', 'de', 'org', 'net', 'info', 'tech']
): Promise<DomainCheckResult[]> {
  const suggestions = await Promise.all(
    tlds.map(tld => checkDomainAvailability(`${domainName}.${tld}`))
  );
  
  return suggestions.filter(s => s.available);
}

export async function registerDomain(
  domain: string,
  customerData: {
    email: string;
    firstName: string;
    lastName: string;
    address: string;
    city: string;
    postalCode: string;
    country: string;
  }
) {
  const response = await fetch(
    `${process.env.RESELLERINTERFACE_API_URL}/domain/register`,
    {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Basic ${Buffer.from(
          `${process.env.RESELLERINTERFACE_API_USERNAME}:${process.env.RESELLERINTERFACE_API_PASSWORD}`
        ).toString('base64')}`
      },
      body: JSON.stringify({
        domain: domain,
        customer: customerData,
        reseller_id: process.env.RESELLERINTERFACE_RESELLER_ID,
        period: 1
      })
    }
  );
  
  return await response.json();
}
```

### 4. API-Route für Domain-Check (app/api/domain/check/route.ts)

```typescript
import { NextRequest, NextResponse } from 'next/server';
import { checkDomainAvailability, getDomainSuggestions } from '@/lib/resellerinterface';

export async function POST(request: NextRequest) {
  try {
    const { domain } = await request.json();
    
    const result = await checkDomainAvailability(domain);
    
    if (!result.available) {
      const domainName = domain.split('.')[0];
      const suggestions = await getDomainSuggestions(domainName);
      result.suggestions = suggestions;
    }
    
    return NextResponse.json(result);
  } catch (error) {
    return NextResponse.json(
      { error: 'Domain-Prüfung fehlgeschlagen' },
      { status: 500 }
    );
  }
}
```

### 5. Frontend-Komponente (components/DomainSearch.tsx)

```typescript
'use client';

import { useState } from 'react';

interface DomainResult {
  available: boolean;
  domain: string;
  price?: number;
  suggestions?: Array<{
    domain: string;
    available: boolean;
    price?: number;
  }>;
}

export default function DomainSearch() {
  const [domain, setDomain] = useState('');
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState<DomainResult | null>(null);

  const handleSearch = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    
    try {
      const response = await fetch('/api/domain/check', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ domain })
      });
      
      const data = await response.json();
      setResult(data);
    } catch (error) {
      console.error('Fehler bei Domain-Suche:', error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-2xl mx-auto p-6">
      <form onSubmit={handleSearch} className="mb-8">
        <div className="flex gap-2">
          <input
            type="text"
            value={domain}
            onChange={(e) => setDomain(e.target.value)}
            placeholder="deinewunschdomain.de"
            className="flex-1 px-4 py-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
            required
          />
          <button
            type="submit"
            disabled={loading}
            className="px-6 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 disabled:bg-gray-400"
          >
            {loading ? 'Prüfe...' : 'Prüfen'}
          </button>
        </div>
      </form>

      {result && (
        <div className="space-y-4">
          <div className={`p-4 rounded-lg ${
            result.available 
              ? 'bg-green-50 border border-green-200' 
              : 'bg-red-50 border border-red-200'
          }`}>
            <h3 className="font-semibold text-lg mb-2">
              {result.available ? '✓ Verfügbar!' : '✗ Bereits vergeben'}
            </h3>
            <p className="text-gray-700">{result.domain}</p>
            {result.available && result.price && (
              <p className="mt-2 text-sm">Preis: {result.price}€/Jahr</p>
            )}
          </div>

          {!result.available && result.suggestions && result.suggestions.length > 0 && (
            <div className="mt-6">
              <h4 className="font-semibold mb-3">Alternative Endungen:</h4>
              <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
                {result.suggestions.map((suggestion) => (
                  <div
                    key={suggestion.domain}
                    className="p-3 bg-white border border-gray-200 rounded-lg hover:border-blue-500 cursor-pointer"
                  >
                    <p className="font-medium">{suggestion.domain}</p>
                    {suggestion.price && (
                      <p className="text-sm text-gray-600 mt-1">
                        {suggestion.price}€/Jahr
                      </p>
                    )}
                    <button className="mt-2 text-sm text-blue-600 hover:underline">
                      Jetzt registrieren
                    </button>
                  </div>
                ))}
              </div>
            </div>
          )}
        </div>
      )}
    </div>
  );
}
```

## B. Kundenverwaltung im ResellerInterface-Portal

### Option 1: Unterkunden-System nutzen (EMPFOHLEN)

Das Unterkunden-System von ResellerInterface bietet professionelle Kundenverwaltung mit individuellen Logins.[4][5][6][7][1]

**Vorteile:**
- Jeder Kunde hat eigenen Login
- Kunden können ihre Domains selbst verwalten
- Automatische Rechnungsstellung
- Separate Preislisten pro Kunde
- Granulares Rechte-Management

### Option 2: Nur Domain-Registrierung ohne Kundenportal

Falls du **nicht** möchtest, dass Kunden eigenes Login haben:

- Registriere Domains direkt unter deinem Haupt-Account
- Speichere Kundenzuordnung in deiner eigenen PostgreSQL-Datenbank
- Verwalte alle Domains zentral
- Biete Kunden-Dashboard auf webwelle.com an

**PostgreSQL-Schema:**

```sql
CREATE TABLE customers (
  id SERIAL PRIMARY KEY,
  email VARCHAR(255) UNIQUE NOT NULL,
  first_name VARCHAR(100),
  last_name VARCHAR(100),
  company VARCHAR(255),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE domains (
  id SERIAL PRIMARY KEY,
  customer_id INTEGER REFERENCES customers(id),
  domain_name VARCHAR(255) UNIQUE NOT NULL,
  resellerinterface_id VARCHAR(255),
  registered_at TIMESTAMP,
  expires_at TIMESTAMP,
  auto_renew BOOLEAN DEFAULT true,
  status VARCHAR(50)
);
```

## Empfehlung für dein Geschäftsmodell

**Option 1 (Unterkunden-System) ist ideal für dich, weil:**

1. **Weniger Support-Aufwand**: Kunden verwalten Domains selbst (DNS-Einträge, Nameserver)[1][4]
2. **Automatische Abrechnung**: ResellerInterface übernimmt Rechnungsstellung[1][7]
3. **Professionelles Erlebnis**: Kunden erhalten eigenes Login-Portal[5][4]
4. **Skalierbarkeit**: System wächst problemlos mit vielen Kunden[2][1]
5. **API-Response < 500ms**: Sehr performant für deine Next.js-Integration[3][2]

**Kombinierter Workflow:**

1. Kunde sucht Domain auf webwelle.com
2. Domain wird über deine API registriert
3. Automatisch wird Unterkunden-Account im ResellerInterface erstellt
4. Kunde erhält Rechnung + Login-Daten
5. Kunde verwaltet Domain selbst im ResellerInterface-Portal

Dies minimiert deinen Verwaltungsaufwand erheblich und bietet Kunden maximale Flexibilität bei der Domain-Verwaltung.erwaltung.[10][1][7][19][4]
